; ================================================================
; MITRA-15 SIMH instruction smoke test
; ================================================================
;
; Memory map used by this test:
;
;   0100-01FF   Test code
;   0200-02FF   Local data segment, L = 0200
;   0040-004F   General/common data, used when G = 0000
;   0180-018F   CDS/context area for CSV/RSV tests
;   0220-0227   Floating-point operands
;   0230-0237   Floating-point/string result area
;   0240-024F   TRS translation table area
;
; Register bases used for most tests:
;
;   L = 0200
;   G = 0000
;
; This makes DL addressing access 0200+displacement.
; DG addressing accesses 0000+displacement.
; LEA with G=0000 returns the effective address directly.
;
; The tests are ordered so that leftover breakpoints from earlier
; groups are not accidentally hit by later shorter test groups.
;
; ================================================================

; ================================================================
; TEST 09: RTS isolated test
; ================================================================
;
; RTS restores P and L from the first two words of the current LDS.
;
; Set L = 0250.
;   (0250) = 0102, so RTS will set P = 0102 + 2 = 0104.
;   (0252) = 0200, so RTS will set L = 0200.
;
; The BRU at 0104 is the halt point.
; ================================================================

deposit cpu L 0250
deposit cpu G 0000

deposit 250 0102
deposit 252 0200

deposit 100 F100     
; RTS
deposit 104 C700     
; BRU $

break 104
run 100

examine cpu P
examine cpu L


; ================================================================
; TEST 10: RSV isolated test
; ================================================================
;
; RSV is privileged and returns from a supervisor section.
; It restores indicators, L, and P from the calling program CDS.
;
; Use G = 0180 as the CDS base.
;
; Formula approximation used here:
;   P = G + 2 + ((G))
;   L = G + ((G+2))
;
; Therefore:
;   ((0180)) = 0104 - 0182 = FF82
;   ((0182)) = 0200 - 0180 = 0080
;
; After RSV, execution should continue at 0104.
; ================================================================

deposit cpu G 0180
deposit cpu MS 1
deposit cpu PR 1

deposit 180 FF82
deposit 182 0080
deposit 184 0000

deposit 100 F10C     
; RSV
deposit 104 C700     
; BRU $

break 104
run 100

examine cpu P
examine cpu L


; ================================================================
; TEST 11: CLS isolated test
; ================================================================
;
; CLS calls a program section.
;
; This test uses:
;   G = 0200
;   section number = 1
;
; The section relocation double word for section 1 is at G-4:
;   0200 - 4 = 01FC
;
; The SRD contains relative L and P values:
;   called L = 0200, relative to G=0200 => 0000
;   called P = 0104, relative to G=0200 => FF04
;
; After CLS, execution should continue at 0104.
; ================================================================

deposit cpu G 0200
deposit cpu L 0200
deposit cpu MS 1

deposit 1FC 0000
deposit 1FE FF04

deposit 100 F801     
; CLS =1
deposit 104 C700     
; BRU $

break 104
run 100

examine cpu P
examine cpu L


; ================================================================
; TEST 12: CSV isolated test
; ================================================================
;
; CSV calls a supervisor section.
;
; The supervisor PRT pointer is stored at absolute address 00C
; hexadecimal, which is decimal 12 in the Mitra-15 manual.
;
; This test uses:
;   PRTS pointer = 0300
;   section number = 1
;
; Supervisor section entry is at pointer - 4:
;   0300 - 4 = 02FC
;
; The entry is loaded as the called section L/P information.
; The exact interpretation may depend on the monitor model, but
; this exercises the CSV opcode and PRTS pointer mechanism.
;
; Use G = 0180 so CSV saves the calling context at 0180.
; ================================================================

deposit cpu G 0180
deposit cpu L 0200
deposit cpu MS 1
deposit cpu PR 1

deposit 00C 0300
deposit 2FC 0200
deposit 2FE 0104

deposit 100 F701     
; CSV =1
deposit 104 C700     
; BRU $

break 104
run 100

examine cpu P
examine cpu L
examine 180
examine 182
examine 184

;

; Reserve area by not depositing meaningful code.

deposit 410 0000
deposit 412 0000
deposit 414 0000
deposit 416 0000

; BND equivalent:
; Merely aligns to a word boundary; no object code.

; EQU, DEF, REF, CDS, LDS, IDS, LPS, FIN, END, PAGE, GOTO, DO,
; and BASE are assembler-time directives and have no direct CPU
; instruction encoding.
