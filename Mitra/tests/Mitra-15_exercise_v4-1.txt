; TEST GROUP 10: Floating Point

echo "TEST 10: Floating-Point"
deposit cpu L 0200
deposit cpu G 0000

; FAD: +1.0 + +1.0 = +2.0
; Function is (E,A)+(Y2,Y2+2)->(E,A): E holds the high (sign+characteristic+
; top mantissa) word, A holds the low mantissa word -- matching how the
; operand was just loaded (LDE from 220, LDA from 222).
; +1.0 = 0x41100000 (C=0x41, M=0x100000). Adding +1.0+1.0 keeps the same
; characteristic (both operands share exponent 16^1) and doubles the
; mantissa: M=0x200000, so +2.0 = 0x41200000 -> E=4120, A=0000.
deposit 4C0 0120
deposit 4C2 0022
deposit 4C4 1A24
deposit 4C6 C700
break 4C6
go 4C0
assert cpu E=4120
assert cpu A=0000
nobreak all

; FSU: +1.0 - +1.0 = 0.0 (a zero mantissa must carry a zero characteristic
; too, per the manual's normalization rule, so both words are all-zero
; regardless of which is "high"/"low").

deposit 4CA 0022
